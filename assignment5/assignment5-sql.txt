SELECT first_orders.order_id, SUM(line_items.quantity * products.price) AS total_price FROM (SELECT order_id FROM orders ORDER BY order_id LIMIT 5) AS first_orders JOIN line_items ON first_orders.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY first_orders.order_id ORDER BY first_orders.order_id;
SELECT customers.customer_name, AVG(order_totals.total_price) AS average_order_price FROM customers JOIN (SELECT orders.customer_id, SUM(line_items.quantity * products.price) AS total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY orders.customer_id, orders.order_id) AS order_totals ON customers.customer_id = order_totals.customer_id GROUP BY customers.customer_id, customers.customer_name ORDER BY customers.customer_name;
